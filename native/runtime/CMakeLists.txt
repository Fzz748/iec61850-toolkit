cmake_minimum_required(VERSION 3.16)
project(iec61850_runtime LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(LIBIEC61850_SOURCE_DIR "" CACHE PATH "Source checkout of libiec61850 v1.6.2.1")
if(NOT EXISTS "${LIBIEC61850_SOURCE_DIR}/src/iec61850/inc/iec61850_server.h")
    message(FATAL_ERROR "Set LIBIEC61850_SOURCE_DIR to libiec61850 v1.6.2.1 sources")
endif()
if(WIN32 AND NOT EXISTS "${LIBIEC61850_SOURCE_DIR}/third_party/winpcap/Lib/wpcap.lib")
    message(FATAL_ERROR "Windows GOOSE requires Npcap SDK headers and x64 import libraries in libiec61850/third_party/winpcap (see docs/runtime.md)")
endif()
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CONFIG_IEC61850_R_GOOSE OFF CACHE BOOL "" FORCE)
set(CONFIG_IEC61850_R_SMV OFF CACHE BOOL "" FORCE)
set(CONFIG_IEC61850_L2_SMV OFF CACHE BOOL "" FORCE)
set(CONFIG_IEC61850_SNTP_CLIENT OFF CACHE BOOL "" FORCE)
set(DEBUG_HAL_ETHERNET OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBIEC61850_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/libiec61850" EXCLUDE_FROM_ALL)
include("${CMAKE_CURRENT_SOURCE_DIR}/goose_events.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/windows_socket.cmake")
add_executable(iec61850-runtime main.c runtime.c json.c platform.c values.c model.c writes.c mms.c goose.c third_party/cjson/cJSON.c)
target_include_directories(iec61850-runtime PRIVATE third_party/cjson
    "${LIBIEC61850_SOURCE_DIR}/hal/inc" "${LIBIEC61850_SOURCE_DIR}/src/common/inc"
    "${LIBIEC61850_SOURCE_DIR}/src/iec61850/inc" "${LIBIEC61850_SOURCE_DIR}/src/mms/inc"
    "${LIBIEC61850_SOURCE_DIR}/src/goose" "${LIBIEC61850_SOURCE_DIR}/src/logging"
    "${LIBIEC61850_SOURCE_DIR}/src/r_session"
    "${CMAKE_BINARY_DIR}/libiec61850/config")
target_link_libraries(iec61850-runtime PRIVATE iec61850 hal)
if(WIN32)
    target_include_directories(iec61850-runtime PRIVATE "${LIBIEC61850_SOURCE_DIR}/third_party/winpcap/Include")
    target_link_libraries(iec61850-runtime PRIVATE ws2_32 iphlpapi bcrypt)
endif()
if(MSVC)
    target_compile_definitions(iec61850-runtime PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(iec61850-runtime PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
    target_link_libraries(iec61850-runtime PRIVATE m)
endif()
install(TARGETS iec61850-runtime RUNTIME DESTINATION bin)
enable_testing()
add_executable(runtime-value-tests EXCLUDE_FROM_ALL test_values.c values.c json.c third_party/cjson/cJSON.c)
get_target_property(runtime_includes iec61850-runtime INCLUDE_DIRECTORIES)
target_include_directories(runtime-value-tests PRIVATE ${runtime_includes})
target_link_libraries(runtime-value-tests PRIVATE iec61850 hal)
if(NOT MSVC)
    target_compile_options(runtime-value-tests PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
    target_link_libraries(runtime-value-tests PRIVATE m)
endif()
add_test(NAME runtime-values COMMAND runtime-value-tests)
